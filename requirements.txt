# =============================================================================
# CONSTANTINE ANTIVIRUS & EDR - DEPENDENCIAS DEL SISTEMA
# =============================================================================
# NOTA ARQUITECTÓNICA FUNDAMENTAL:
# Constantine contempla integraciones con herramientas líderes de ciberseguridad
# (Cilium Tetragon, CNCF Falco, Wazuh, osquery, MISP, Sigma y ClamAV).
#
# Algunas de estas integraciones son bibliotecas de Python instalables vía pip,
# mientras que otras corresponden a SERVICIOS DEL SISTEMA OPERATIVO O DEMONIOS
# INDEPENDIENTES que no se instalan únicamente mediante pip.
#
# Si un demonio o paquete externo no está instalado, Constantine activa
# automáticamente su motor standalone nativo o modo de degradación asistida,
# garantizando que el antivirus arranque y opere sin colapsar.
# =============================================================================

# -----------------------------------------------------------------------------
# 1. DEPENDENCIAS OBLIGATORIAS DEL NÚCLEO (Core Mandatory)
# -----------------------------------------------------------------------------
# Requerido para cifrado autenticado AES-256-GCM del Baúl Criptográfico y gestión de claves
cryptography>=38.0.0

# Requerido para parseo de reglas de comportamiento (Sigma, Falco y Tetragon TracingPolicies)
PyYAML>=6.0

# Requerido para la interfaz gráfica nativa de escritorio GTK 3.0 (vía PyGObject)
# Nota: En distribuciones basadas en Debian/Ubuntu se recomienda instalar 'python3-gi' vía APT
PyGObject>=3.42.0

# -----------------------------------------------------------------------------
# 2. INTEGRACIONES PYTHON OPCIONALES (Optional Python Libraries)
# -----------------------------------------------------------------------------
# Adaptador oficial de reglas Sigma para compilación de AST y backends específicos
# Si no está instalado, Constantine utiliza su motor de evaluación YAML standalone interno
pySigma>=0.11.0; extra == "sigma"

# Cliente oficial de la API REST de MISP para sincronización bidireccional de IoCs
# Si no está instalado, Constantine consume feeds e indicadores locales en formato JSON/CSV
pymisp>=2.4.170; extra == "misp"

# Bindings nativos oficiales de YARA (C-YARA)
# Si no está disponible, Constantine compila y ejecuta mediante el motor autónomo YaraXEngine
yara-python>=4.3.0; extra == "yara"

# Conector Python para el socket Unix del demonio clamd de ClamAV
# Si no está instalado, Constantine se comunica directamente mediante sockets Unix nativos
clamd>=1.0.2; extra == "clamav"

# -----------------------------------------------------------------------------
# 3. DEMONIOS Y SERVICIOS DEL SISTEMA OPERATIVO (Independent System Services)
# -----------------------------------------------------------------------------
# Los siguientes componentes son programas independientes del sistema. NO se
# pueden instalar simplemente mediante 'pip install'. Para habilitar el modo
# LIVE_SERVICE completo en lugar de STANDALONE_EMULATION, instale los paquetes
# correspondientes en su distribución Linux:
#
# • ClamAV Antivirus Daemon:
#   Debian/Ubuntu: sudo apt-get install clamav clamav-daemon
#   RHEL/Fedora:   sudo dnf install clamav clamd
#
# • osquery (Diagnóstico e Inventario SQL):
#   Descarga oficial / repositorio: https://osquery.io
#   Paquetes: osqueryi, osqueryd (Socket: /var/osquery/osquery.em)
#
# • CNCF Falco (Observabilidad Conductual de Syscalls):
#   Descarga oficial: https://falco.org
#   Paquete: falco (Socket gRPC: /run/falco/falco.sock)
#
# • Wazuh Agent (Integridad de Ficheros FIM & Respuesta Activa):
#   Descarga oficial: https://wazuh.com
#   Paquete: wazuh-agent (Directorio: /var/ossec)
#
# • Cilium Tetragon (Seguridad en Tiempo de Ejecución eBPF):
#   Descarga oficial: https://github.com/cilium/tetragon
#   Paquete: tetragon (Socket: /var/run/cilium/tetragon/tetragon.sock)
#
# • AIDE (Verificación Criptográfica de Integridad del Sistema):
#   Debian/Ubuntu: sudo apt-get install aide aide-common
#   RHEL/Fedora:   sudo dnf install aide
#
# • Cortafuegos de Escritorio:
#   Debian/Ubuntu: sudo apt-get install gufw ufw
# -----------------------------------------------------------------------------
